26ekf = EKF()
27myIMU = IMU()

30def sensor_stream():
31 # For testing purposes
32 # This creates a CSV file that will store the raw data and the EKF predictions for each loop iteration
33 timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
34 os.makedirs('output', exist_ok=True)
35 path = f"output/output_{timestamp}.csv"
36 count = 1
37
38 with open(path, "w", newline="", encoding="utf-8") as f:
39 cWriter = csv.writer(f)
40 cWriter.writerow(["id", "latitude", "longitude", "speed", "heading",
41 "gps-lat", "gps-long", "dt", "imu-ax", "imu-ay", "imu-az",
42 "imu-gx", "imu-gy", "imu-gz", "imu-mx", "imu-my", "imu-mz", ])
43
44 print(f"Saving to: {path}")
45
46 yaw_offset = 0.0 # Yaw = heading
47 yaw_offset_initialized = False
48
49 prev_gps_x = None
50 prev_gps_y = None
51 prev_gps_time = None
52
53 prev_time = time.time()
54
55 # Initialize GPS origin
56 # Ensures that the GPS has locked onto a point before initializing the origin
57 # Prevents false points like (0,0)
58 while True:
59 lat_init, lon_init = myGPS.get_lat(), myGPS.get_long()
60 if lat_init != 0 and lon_init != 0:
61 break
62 time.sleep(0.1)
63
64 last_gps_time = 0
65 last_mag_time = 0
66
67 # accel_measured = accel_true + bias + noise
68 # When car is not moving, accel_true is 0 and there is no noise...so that means any accel_measured = bias
69 bias_samples = []
70 for _ in range(100):
71 a = myIMU.get_accel()
72 bias_samples.append(a[0])
73 time.sleep(0.01)
74
75 accel_bias_x = np.mean(bias_samples)
76 print(f"Estimated accel bias x: {accel_bias_x}") # Car needs to remain still until this message pops up in terminal
77
78 while True:
79 current_time = time.time()
80 dt = current_time - prev_time
81 prev_time = current_time
82
83 dt = min(dt, 0.02) # Cap dt to 20 ms to prevent large jumps (dt clamp)
84
85 # This is to ensure that consistent values are used for all updates within a single iteration of the loop
86 accel = myIMU.get_accel()
87 gyro = myIMU.get_gyro()
88 mag = myIMU.get_magn()
89 lat = myGPS.get_lat()
90 lon = myGPS.get_long()
91
92 # Project acceleration onto heading direction
93 # The logic here is that the car is either traveling forwards or backwards, even while turning it is still technically
94 # maintaining acceleration in the direction that the x-direction of the IMU is facing. The car does not travel sideways,
95 # so we do not need accel[1]
96 a_forward = accel[0] - accel_bias_x
97 a_forward = np.clip(a_forward, -5.0, 5.0) # Prevents bad numbers from hurting the EKF predictions
98 omega = gyro[2]
99 omega = np.clip(omega, -3.0, 3.0) # Prevents bad numbers from hurting the EKF predictions
100 u = np.array([[a_forward], [omega]])
101
102 # EKF prediction
103 ekf.predict(u, dt)
104
105 # GPS update
106 if current_time - last_gps_time > 0.2: # GPS has an ODR of 5 Hz
107 gps_x, gps_y = latlon_toxy(lat, lon, lat_init, lon_init) # Converts lat and long into Cartesian coordinates b/c IMU measures in meters and typical GPS coords are angular
108 z_gps = np.array([gps_x, gps_y])
109
110 # GPS gate to prevent bad numbers from hurting EKF predictions
111 dist_to_state = np.linalg.norm([
112 gps_x - ekf.x[0, 0],
113 gps_y - ekf.x[1, 0]
114 ])
115
116 old_R = ekf.R_gps.copy()
117
118 if dist_to_state < 25.0:
119 ekf.update_gps(np.array([gps_x, gps_y]))
120 elif dist_to_state < 100.0:
121 ekf.R_gps = np.diag([400.0, 400.0])
122 ekf.update_gps(z_gps)
123 ekf.R_gps = old_R
124
125 # Auto-calibrate yaw offset using GPS heading
126 if prev_gps_x is not None and prev_gps_time is not None:
127 dx = gps_x - prev_gps_x
128 dy = gps_y - prev_gps_y
129 dist = np.hypot(dx, dy)
130 gps_dt = current_time - prev_gps_time
131
132 # Only calibrates when GPS movement is big enough to give a meaningful heading, when the user is
133 # traveling roughly straight, and updates slowly so noise does not jerk the heading around
134 if gps_dt > 0 and dist > 3.0 and abs(gyro[2]) < 0.2:
135 gps_heading = np.arctan2(dy, dx)
136 mag_yaw_raw = tilt_compensated_yaw(accel, mag)
137
138 if mag_yaw_raw is not None and np.isfinite(mag_yaw_raw):
139 candidate_offset = ekf.normalize_angle(gps_heading - mag_yaw_raw)
140
141 if not yaw_offset_initialized:
142 yaw_offset = candidate_offset
143 yaw_offset_initialized = True
144 else:
145 yaw_offset = blend_angle(ekf, yaw_offset, candidate_offset, alpha=0.05)
146
147 prev_gps_x = gps_x
148 prev_gps_y = gps_y
149 prev_gps_time = current_time
150 last_gps_time = current_time
151
152 # Magnetometer update
153 if current_time - last_mag_time >= 0.01: # Magnetometer ODR is 100 Hz
154 mag_yaw = tilt_compensated_yaw(accel, mag) # THIS IS IN RADIANS
155 if mag_yaw is not None: # Checks for invalid input
156 mag_yaw = ekf.normalize_angle(mag_yaw + np.pi + yaw_offset)
157 ekf.update_yaw(mag_yaw)
158 last_mag_time = current_time
159
160 # velocity clamp, essentially if the filter believes the user is barely moving, it forces speed down to 0
161 if abs(ekf.x[2, 0]) < 0.02:
162 ekf.x[2, 0] = 0
163
164 # Extract state
165 x, y, v, mag_yaw = ekf.x.flatten()
166 # print(f"x={x:.2f}, y={y:.2f}, v={v:.2f}, mag_yaw={np.degrees(mag_yaw):.1f} rad")
167 #R = 6378137 # Earth radius
168 #lat_est = lat_init + (y / R) * (180 / np.pi)
169 #lon_est = lon_init + (x / (R * np.cos(np.radians(lat_init)))) * (180 / np.pi)
170
171 # Saves state in testing output CSV file
172 with open(path, "a", newline="", encoding="utf-8") as f:
173 cWriter = csv.writer(f)
174 cWriter.writerow([count, x, y, v, mag_yaw, lat, lon, dt,
175 accel[0], accel[1], accel[2],
176 gyro[0], gyro[1], gyro[2],
177 mag[0], mag[1], mag[2]])
178 count += 1
179
180 data = {
181 "x": float(x),
182 "y": float(y),
183 "v": float(v),
184 "mag_yaw": float(mag_yaw),
185 "lat": float(lat),
186 "lon": float(lon)
187 }
188
189 # SSE format
190 yield f"data: {json.dumps(data)}\n\n"
191
192 time.sleep(0.01) # Sleep to prevent busy loop, adjusted to 0.01 to match the fastest ODR (magnetometer ODR 100Hz)
193
194
195@app.route('/stream')
196def stream():
197 return Response(sensor_stream(), mimetype='text/event-stream',
198 headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no"}) # important for streaming
